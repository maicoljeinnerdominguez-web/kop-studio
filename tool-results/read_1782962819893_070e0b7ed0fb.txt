   127→function ImageZoom({
   128→  src,
   129→  alt,
   130→  isSelected,
   131→}: {
   132→  src: string
   133→  alt: string
   134→  isSelected: boolean
   135→}) {
   136→  const containerRef = useRef<HTMLDivElement>(null)
   137→  const [isZoomed, setIsZoomed] = useState(false)
   138→  const [zoomPosition, setZoomPosition] = useState({ x: 50, y: 50 })
   139→
   140→  const handleMouseMove = useCallback(
   141→    (e: React.MouseEvent<HTMLDivElement>) => {
   142→      if (!containerRef.current) return
   143→      const rect = containerRef.current.getBoundingClientRect()
   144→      const x = ((e.clientX - rect.left) / rect.width) * 100
   145→      const y = ((e.clientY - rect.top) / rect.height) * 100
   146→      setZoomPosition({ x, y })
   147→    },
   148→    []
   149→  )
   150→
   151→  const handleMouseEnter = useCallback(() => setIsZoomed(true), [])
   152→  const handleMouseLeave = useCallback(() => setIsZoomed(false), [])
   153→
   154→  if (!isSelected) return null
   155→
   156→  return (
   157→    <div
   158→      ref={containerRef}
   159→      className="w-full h-full overflow-hidden cursor-zoom-in"
   160→      onMouseMove={handleMouseMove}
   161→      onMouseEnter={handleMouseEnter}
   162→      onMouseLeave={handleMouseLeave}
   163→    >
   164→      <motion.img
   165→        src={src}
   166→        alt={alt}
   167→        initial={{ opacity: 0 }}
   168→        animate={{ opacity: 1 }}
   169→        transition={{ duration: 0.3 }}
   170→        className="w-full h-full object-cover transition-transform duration-200 ease-out img-blur-up"
   171→        style={
   172→          isZoomed
   173→            ? {
   174→                transform: 'scale(1.5)',
   175→                transformOrigin: `${zoomPosition.x}% ${zoomPosition.y}%`,
   176→              }
   177→            : {
   178→                transform: 'scale(1)',
   179→              }
   180→        }
   181→      />
   182→    </div>
   183→  )
   184→}
